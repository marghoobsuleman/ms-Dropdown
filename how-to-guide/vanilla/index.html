ms-Dropdown Vanilla JavaScript Example

Basic Example


// JavaScript
const dropdown = new MsDropdown('#basicDropdown', {
  showFilterAlways: true,
  visibleRows: 5
});

// Access properties
console.log(dropdown.value);

// Call methods
dropdown.refresh();

Multiple Selection


// JavaScript
const multiDropdown = new MsDropdown('#multipleDropdown', {
  enableCheckbox: true
});

// Get multiple values
console.log(multiDropdown.value); // Returns array of selected values

Creating from JSON


// JavaScript
function createFromJson() {
  const jsonData = [
    { text: 'Facebook', value: 'fb', image: '../../dist/images/icons/icon_email.gif' },
    { text: 'Twitter', value: 'tw', image: '../../dist/images/icons/icon_faq.gif' },
    { text: 'LinkedIn', value: 'li', image: '../../dist/images/icons/icon_games.gif' }
  ];

  MsDropdown.make('#jsonDropdown', {
    byJson: {
      data: jsonData,
      selectedIndex: 0
    }
  });
}

Adding Options Dynamically


// JavaScript
const dynamicDropdown = new MsDropdown('#dynamicDropdown');

function addOption() {
  dynamicDropdown.add({
    text: 'Banana',
    value: 'banana',
    image: '../../dist/images/icons/icon_search.gif'
  });
}
