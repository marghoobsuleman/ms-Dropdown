ms-Dropdown TypeScript Examples

This page demonstrates how to use ms-Dropdown with TypeScript in various scenarios.

Basic Usage with TypeScript

TypeScript
HTML
Demo
// Import the MsDropdown class and its types
import MsDropdown, { MsDropdownOptions } from 'ms-dropdown';

// Define options with type safety
const options: MsDropdownOptions = {
  showFilterAlways: true,
  visibleRows: 5,
  imagePosition: 'left'
};

// Create a dropdown with type checking
const dropdown = new MsDropdown('#basicDropdown', options);

// Type-safe property access
const value: string = dropdown.value;
console.log(`Selected value: ${value}`);

// Type-safe method calls
dropdown.showRows(8);

// Add a new option with type checking
dropdown.add({
  text: 'New Option',
  value: 'new',
  image: 'path/to/image.png',
  description: 'This is a new option'
});

</select>

React with TypeScript

TypeScript
TSX
// Type definitions for props
import React, { useState } from 'react';
import MsDropdown, { MsDropdownOptionProps } from 'ms-dropdown/react';

// Type-safe option definition
const socialOptions: MsDropdownOptionProps[] = [
  { text: 'Facebook', value: 'fb', image: './images/fb.gif' },
  { text: 'Twitter', value: 'tw', image: './images/tw.gif' },
  { text: 'LinkedIn', value: 'li', image: './images/in.gif' }
];

// Component with TypeScript
const SocialDropdown: React.FC = () => {
  const [selected, setSelected] = useState<string>('');

  const handleChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    setSelected(e.target.value);
    console.log(`Selected: ${e.target.value}`);
  };

  return (
    <div>
      <h3>Select a Social Network</h3>
      <MsDropdown
        options={socialOptions}
        value={selected}
        onChange={handleChange}
        imagePosition="left"
        showFilterAlways={false}
      />
      <p>Selected value: {selected}</p>
    </div>
  );
};
// App.tsx
import React from 'react';
import SocialDropdown from './SocialDropdown';

const App: React.FC = () => {
  return (
    <div className="app">
      <h1>ms-Dropdown React TypeScript Example</h1>
      <SocialDropdown />
    </div>
  );
};

export default App;

Vue with TypeScript

TypeScript
Template
// SocialDropdown.vue
<script lang="ts">
import { defineComponent, ref } from 'vue';
import MsDropdown from 'ms-dropdown/vue';

interface SocialOption {
  text: string;
  value: string;
  image?: string;
  description?: string;
}

export default defineComponent({
  components: {
    MsDropdown
  },
  setup() {
    const selected = ref('');
    
    // Type-safe options array
    const options = ref<SocialOption[]>([
      { text: 'Facebook', value: 'fb', image: './images/fb.gif' },
      { text: 'Twitter', value: 'tw', image: './images/tw.gif' },
      { text: 'LinkedIn', value: 'li', image: './images/in.gif' }
    ]);
    
    const handleChange = (value: string) => {
      selected.value = value;
      console.log(`Selected: ${value}`);
    };
    
    return {
      selected,
      options,
      handleChange
    };
  }
});
</script>
<template>
  <div>
    <h3>Select a Social Network</h3>
    <MsDropdown
      v-model="selected"
      :options="options"
      imagePosition="left"
      :showFilterAlways="false"
      @update:modelValue="handleChange"
    />
    <p>Selected value: {{ selected }}</p>
  </div>
</template>

Bootstrap with TypeScript

// Import the Bootstrap integration with TypeScript
import MsDropdownBootstrap from 'ms-dropdown/bootstrap';

// Define options with TypeScript interface
interface BootstrapOptions {
  className?: string;
  showFilterAlways?: boolean;
  imagePosition?: string;
}

// Create options with type safety
const options: BootstrapOptions = {
  className: 'form-select-lg',
  showFilterAlways: true
};

// Initialize specific dropdown with type checking
const dropdown = new MsDropdownBootstrap(
  document.querySelector('#bootstrapDropdown') as HTMLElement,
  options
);

// Or initialize all dropdowns with a selector
const allDropdowns = MsDropdownBootstrap.init('select.ms-dropdown');

// Type-safe method calls
dropdown.refresh();

// Event handling with TypeScript
document.querySelector('#refreshButton')?.addEventListener('click', () => {
  dropdown.refresh();
});
